@echo off
chcp 65001 >nul
title AQUA-8 - Iniciando Sistema...
color 0B

echo ============================================
echo    AQUA-8 - Capitulos de Ingenieria
echo    8AMPERIOS SAS
echo ============================================
echo.

REM Verificar que existen las carpetas
if not exist "C:\Users\ASUS}\Desktop\RELX\backend" (
    echo [ERROR] No se encontro la carpeta backend
    pause
    exit /b 1
)

if not exist "C:\Users\ASUS}\Desktop\RELX\frontend" (
    echo [ERROR] No se encontro la carpeta frontend
    pause
    exit /b 1
)

echo [1/4] Verificando entorno virtual backend...
cd /d "C:\Users\ASUS}\Desktop\RELX\backend"
if not exist "venv\Scripts\activate.bat" (
    echo [INFO] Creando entorno virtual...
    python -m venv venv
)

echo [2/4] Activando entorno virtual...
call venv\Scripts\activate.bat

echo [3/4] Verificando dependencias backend...
pip show django >nul 2>&1
if errorlevel 1 (
    echo [INFO] Instalando dependencias backend...
    pip install -r requirements.txt
)

echo [4/4] Verificando base de datos...
if not exist "db.sqlite3" (
    echo [INFO] Creando base de datos...
    python manage.py migrate
    echo [INFO] Creando superusuario...
    python manage.py createsuperuser
    echo [INFO] Cargando capitulos maestros...
    python manage.py shell < ..\scripts\init_db.py
)

echo.
echo ============================================
echo    Iniciando servicios...
echo ============================================
echo.

REM Abrir backend en nueva ventana
start "AQUA-8 BACKEND" cmd /k "cd /d C:\Users\ASUS}\Desktop\RELX\backend && venv\Scripts\activate.bat && python manage.py runserver"

timeout /t 3 /nobreak >nul

REM Abrir frontend en nueva ventana
start "AQUA-8 FRONTEND" cmd /k "cd /d C:\Users\ASUS}\Desktop\RELX\frontend && npm run dev"

timeout /t 3 /nobreak >nul

echo [OK] Backend iniciado en http://localhost:8000
echo [OK] Frontend iniciado en http://localhost:5173
echo.
echo Abriendo navegador...
start http://localhost:5174/

echo.
echo ============================================
echo    AQUA-8 esta corriendo!
echo    Backend:  http://localhost:8000
echo    Frontend: http://localhost:5174
echo    Admin:    http://localhost:8000/admin
echo ============================================
echo.
echo Presiona cualquier tecla para cerrar esta ventana...
pause >nul
